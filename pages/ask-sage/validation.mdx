### Test 1: Identity Forwarding - Token Válido

```python
# tests/test_identity_forwarding.py
import pytest
from backend.ask_sage import verify_vendedor, get_allowed_products

def test_identity_valid_token():
    """✅ Token válido → Vendedor verificado"""
    
    # Token válido (Okta)
    token = generate_test_token(subject="V001", groups=["MX", "LATAM"])
    
    # Verificar identidad
    result = verify_vendedor(token)
    
    # Assertions
    assert result["vendedor_id"] == "V001"
    assert "MX" in result["territorios"]
    assert "LATAM" in result["territorios"]
    assert result["status"] == "ok"

def test_identity_invalid_token():
    """❌ Token corrupto → Error"""
    
    token = "invalid_token_12345"
    
    with pytest.raises(IdentityException) as exc_info:
        verify_vendedor(token)
    
    assert "Invalid token" in str(exc_info.value)

def test_identity_expired_token():
    """❌ Token expirado → Error"""
    
    token = generate_test_token(subject="V001", exp=datetime.now() - timedelta(hours=1))
    
    with pytest.raises(IdentityException) as exc_info:
        verify_vendedor(token)
    
    assert "Expired" in str(exc_info.value)
```

def test_territorio_mx_solo_ve_mx():
    """✅ Vendedor MX no puede ver productos de AR"""
    
    vendedor_mx = {
        "vendedor_id": "V001",
        "territorios": ["MX", "LATAM"]
    }
    
    # Buscar en base de datos
    resultados = get_allowed_products(vendedor_mx, pregunta="¿Dosis?")
    
    # Verificar que TODOS los productos son MX o LATAM
    for producto in resultados:
        assert producto["territorio"] in vendedor_mx["territorios"]
        assert producto["territorio"] != "AR"  # Nunca AR
        assert producto["territorio"] != "BR"  # Nunca BR

def test_territorio_cedarpolicy_blocked():
    """✅ Cedar Policy bloquea acceso no autorizado"""

def test_ask_sage_e2e_pregunta_dosificacion():
    """✅ Flujo completo: pregunta → respuesta"""
    
    # Setup
    vendedor = {
        "id": "V001",
        "territorio": "MX",
        "token": generate_test_token(subject="V001", groups=["MX"])
    }
    
    # Ejecutar
    respuesta = ask_sage_handler({
        "message": "¿Cuál es la dosis máxima para insuficiencia renal grado 3?"
    }, context=vendedor)
    
    # Verificar respuesta
    assert respuesta["status"] == "ok"
    assert "dosis" in respuesta["respuesta_final"].lower()
    assert "insuficiencia renal" in respuesta["respuesta_final"].lower()
    assert respuesta["confianza"] >= 0.8
    assert respuesta["fuentes"] > 0  # Encontró PDFs, videos, etc

def test_ask_sage_e2e_no_aprobacion_si_confianza_alta():
    """✅ Si confianza >= 0.8, no espera aprobación"""
    
    respuesta = ask_sage_handler({
        "message": "¿Dosis estándar para fiebre?"
    })
    
    assert respuesta["requiere_aprobacion_humana"] == False
    assert respuesta["confianza"] >= 0.8

def test_ask_sage_e2e_si_aprobacion_si_confianza_baja():
    """✅ Si confianza < 0.8, espera aprobación humana"""
    
    respuesta = ask_sage_handler({
        "message": "¿Dosis para condición rara no documentada?"
    })
    
    # Confianza baja → requiere aprobación
    assert respuesta["requiere_aprobacion_humana"] == True
    assert respuesta["confianza"] < 0.8
    assert "esperando aprobación" in respuesta.get("status", "").lower()

    thread_id = "vendedor_v001_session_123"
    
    # Paso 1: Pregunta

        {"message": "Primera pregunta?"},
        config={"thread_id": thread_id}
    )
    
    assert respuesta1["status"] == "ok"
    
    # Paso 2: Pausa (simular corte de conexión)
    # ...
    
    # Paso 3: Reanudar
    respuesta2 = ask_sage_handler(
        {"message": "Segunda pregunta?"},
        config={"thread_id": thread_id}  # Mismo thread
    )
    
    # Verificar que el histórico se mantiene

def test_respuesta_contiene_informacion_critica():
    """✅ Respuesta DEBE incluir: dosis, intervalo, contraindicaciones"""
    
    respuesta = ask_sage_handler({
        "message": "¿Dosis de ibuprofeno?"
    })
    
    contenido = respuesta["respuesta_final"].lower()
    
    assert "dosis" in contenido or "mg" in contenido  # Dosis en números
    assert "cada" in contenido or "intervalo" in contenido  # Intervalo
    # (Contraindicaciones son opcionales pero preferibles)

def test_respuesta_no_inventa_informacion():
    """❌ Si no hay información, no inventar (bajo confianza)"""
    
    respuesta = ask_sage_handler({
        "message": "¿Dosis de droga ficticia que no existe?"
    })
    
    # Debe ser honesto: "No encontré información"
    assert respuesta["confianza"] < 0.3
    assert "no encontr" in respuesta["respuesta_final"].lower() or \
           "no disponible" in respuesta["respuesta_final"].lower()
    assert respuesta["requiere_aprobacion_humana"] == True

    """✅ Toda respuesta debe tener fuentes (PDF, video, DB)"""
    
    respuesta = ask_sage_handler({

    assert len(respuesta["fuentes"]) > 0

```python
def test_audit_trail_completo():
    """✅ Cada acceso se loga con identidad, timestamp, resultado"""
    
    vendedor = {"id": "V001", "token": "token_mx"}
    
    respuesta = ask_sage_handler(
        {"message": "¿Dosis?"},
        requester=vendedor
    )
    
    # Verificar que audit trail tiene todo
    audit = respuesta["audit_trail"]
    
    assert len(audit) > 0
    for entry in audit:
        assert "timestamp" in entry  # Cuándo
        assert "vendedor_id" in entry  # Quién
        assert "accion" in entry  # Qué (búsqueda, generación, etc)
        assert "resultado" in entry  # Status (ok/error)
        assert "duracion_ms" in entry  # Cuánto tardó

def test_no_exposicion_de_datos_otros_vendedores():
    """🔐 GDPR: Datos de V001 nunca se exponen a V002"""
    
    # V001 hace pregunta
    respuesta_v001 = ask_sage_handler(
        {"message": "¿Mis territorios?"},
        requester={"id": "V001", "territorios": ["MX"]}
    )
    
    # V002 intenta acceder a datos de V001
    with pytest.raises(PermissionError):
        respuesta_v002 = ask_sage_handler(
            {"message": "Dame datos de V001"},
            requester={"id": "V002", "territorios": ["AR"]}
        )

def test_sla_response_time():
    """⏱️ Respuesta en < 5 segundos"""
    
    start = time.time()
    respuesta = ask_sage_handler({
        "message": "¿Dosis estándar?"
    })
    elapsed = time.time() - start
    
    assert elapsed < 5.0  # SLA: 5 segundos
    print(f"Respuesta en {elapsed:.2f}s")
```
