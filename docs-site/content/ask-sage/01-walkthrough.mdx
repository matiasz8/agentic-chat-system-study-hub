```javascript
// pages/chat/[slug].tsx
export default function ChatPage() {
  const [messages, setMessages] = useState([]);

  const handleSubmit = async (question: string) => {
    // ✅ El vendedor pregunta
    // ✅ Su token Okta se adjunta automáticamente (interceptor HTTP)

    const response = await fetch('/api/chat', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${okta_token}`, // 🔐
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        question,
        context: { vendedor_id: 'V001', territorio: 'MX' }
      })
    });

    // ✅ Streaming: recibir tokens uno por uno
    const stream = response.body.getReader();
    const decoder = new TextDecoder();

    while (true) {
      const { done, value } = await stream.read();
      if (done) break;

      const chunk = decoder.decode(value);
      // 🎬 Mostrar cada token en tiempo real
      setMessages(prev => [...prev, chunk]);
    }
  };
}
```

**Conceptos Ask Sage**:
- ✅ Streaming: Vendedor ve "Por favor espere" → "Buscando" → "Encontré" (en tiempo real)
- ✅ Generative UI: Tabla de dosis como componente React (no texto plano)

from aws_agentcore import verify_identity, get_allowed_resources

async def verify_vendedor_access(token: str, territorio: str):
    """
    ✅ Verifica: ¿Puede este vendedor acceder a datos de este territorio?
    """

    # 1. Validar token Okta
    claims = verify_identity(token)  # → {'sub': 'V001', 'groups': ['MX', 'LATAM']}

    # 2. Buscar recursos autorizados
    allowed = get_allowed_resources(
        user_id=claims['sub'],
        territory=territorio
    )

    # 3. Ejemplo de respuesta:
    return {
        'vendedor_id': 'V001',
        'territorio': 'MX',
        'allowed_products': ['P001', 'P002', 'P003'],  # ✅ Solo estos
        'allowed_territories': ['MX', 'LATAM'],
        'policies': ['ReadOnly', 'NoExport']
    }

```python
# backend/orchestration/ask_sage_graph.py
from langgraph.graph import StateGraph
from typing_extensions import TypedDict

class AskSageState(TypedDict):
    """Estado centralizado de Ask Sage"""
    vendedor_id: str
    territorio: str
    pregunta: str
    contexto_permitido: list  # ✅ Productos permitidos
    resultado_busqueda: dict
    respuesta_generada: str
    confianza: float
    requiere_aprobacion: bool

def create_ask_sage_graph():
    graph = StateGraph(AskSageState)

    # NODOS: Funciones ejecutables
    def nodo_buscar_documentos(state: AskSageState) -> AskSageState:
        """Buscar en PDFs, videos, DB"""
        # 🔍 Busca respuesta usando MCP
        estado['resultado_busqueda'] = {
            'fuente': 'PDF Bula Técnica pág 12',
            'contenido': 'Dosis máxima: 100mg/día',
            'confianza': 0.95
        }
        return state

    def nodo_generar_respuesta(state: AskSageState) -> AskSageState:
        """Generar respuesta final con componentes React"""
        state['respuesta_generada'] = f"""
        Basado en {state['resultado_busqueda']['fuente']}:

        **Dosis máxima para insuficiencia renal grado 3: 100mg/día**

        [TABLA INTERACTIVA GENERADA AQUÍ]
        """
        state['confianza'] = state['resultado_busqueda']['confianza']
        return state

    def nodo_validacion(state: AskSageState) -> AskSageState:
        """¿Necesita aprobación humana?"""
        state['requiere_aprobacion'] = state['confianza'] < 0.8
        return state

    # Agregar nodos
    graph.add_node("buscar", nodo_buscar_documentos)
    graph.add_node("generar", nodo_generar_respuesta)
    graph.add_node("validar", nodo_validacion)

    # ARISTAS: Rutas condicionales
    graph.add_edge("buscar", "generar")

    # Condicional: ¿Confianza >= 0.8?
    def arista_confianza(state: AskSageState):
        if state['confianza'] >= 0.8:
            return "RESPUESTA_LISTA"
        else:
            return "APROBACION_HUMANA"

    graph.add_conditional_edges(
        "validar",
        arista_confianza,
        {
            "RESPUESTA_LISTA": "__end__",
            "APROBACION_HUMANA": "esperar_aprobacion"
        }
    )

    return graph.compile()

# Ejecutar
ask_sage_graph = create_ask_sage_graph()
resultado = ask_sage_graph.invoke(initial_state)
```

**Conceptos Ask Sage**:
- ✅ State: Guardar contexto (qué búscó, qué encontró)
- ✅ Nodos: Funciones independientes (buscar, generar, validar)
- ✅ Aristas: Decidir siguientes pasos (if confianza >= 0.8)

### PASO 4: DATA LAYER - Buscar Información

**Ubicación**: `data/30_mcp_protocol` + `data/31_multimodal_embeddings`

```python
# backend/data/mcp_client.py
from mcp import MCPClient
from embeddings import MultimodalEmbedding

class AskSageDataLayer:
    """Acceso unificado a PDFs, DB, Videos"""

    def __init__(self):
        self.mcp = MCPClient()
        self.embeddings = MultimodalEmbedding()

    def buscar_respuesta(self, pregunta: str, contexto: dict) -> dict:
        """
        Busca en:
        1. PDFs (embeddings de texto)
        2. DB (queries SQL)
        3. Videos (embeddings de frames)
        """

        # 1. Vectorizar pregunta
        vector_pregunta = self.embeddings.embed_text(pregunta)

        # 2. Buscar similares en Pinecone
        resultados_pdf = self.mcp.search(
            vector=vector_pregunta,
            source="pdfs",
            territorio=contexto['territorio']  # ✅ Solo datos permitidos
        )

        # 3. Query a base de datos
        resultados_db = self.mcp.query(
            """
            SELECT * FROM farmaco_dosis
            WHERE condicion LIKE %s
            AND territorio IN %s
            """,
            params=("insuficiencia renal", contexto['territories'])
        )

        # 4. Buscar en videos
        resultados_videos = self.mcp.search_videos(
            query="dosificación insuficiencia renal",
            territorio=contexto['territorio']
        )

        return {
            'pdfs': resultados_pdf,
            'db': resultados_db,
            'videos': resultados_videos,
            'confianza': max([r['score'] for r in resultados_pdf])
        }

# Resultado:
# {
#   'pdfs': [{'fuente': 'Bula.pdf', 'pag': 12, 'score': 0.95}],
#   'db': [{'dosis': '100mg', 'territorio': 'MX'}],
#   'videos': [{'url': 'youtube...', 'timestamp': '4:32'}],
#   'confianza': 0.95
# }
```

**Conceptos Ask Sage**:
- ✅ MCP: Acceso uniforme a cualquier fuente (PDF, DB, Video)
- ✅ Embeddings: Buscar por similitud semántica
- ✅ Multimodal: Procesar texto + imágenes + video
